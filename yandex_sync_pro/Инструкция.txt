# 1. Создайте виртуальное окружение (рекомендуется)
python -m venv venv
venv\Scripts\activate

# 2. Установка всех зависимостей
pip install -r requirements.txt --upgrade

# 3. Для Windows - дополнительные пакеты автозапуска
pip install pywin32 winshell

# 4. Запуск приложения
python main.py

# 5. Сборка в один EXE файл (опционально)
pyinstaller --onefile --windowed --icon=assets/icon.ico ^
  --name yandex_sync_pro ^
  --add-data "assets;assets" ^
  main.py

# 6. Сборка инсталлятора (требуется Inno Setup)
#    - Установите Inno Setup (https://jrsoftware.org/isdl.php)
#    - Откройте installer.iss в компиляторе
#    - Build → Compile
