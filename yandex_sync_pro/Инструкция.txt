# 1. Установка всех зависимостей
pip install -r requirements.txt

# 2. Для Windows - дополнительные пакеты автозапуска
pip install pywin32 winshell

# 3. Запуск приложения
python main.py

# 4. Сборка в один EXE файл (опционально)
pyinstaller --onefile --windowed --icon=assets/icon.ico ^
  --name yandex_sync_pro ^
  --add-data "assets;assets" ^
  main.py

# 5. Сборка инсталлятора (требуется Inno Setup)
#    - Установите Inno Setup (https://jrsoftware.org/isdl.php)
#    - Откройте installer.iss в компиляторе
#    - Build → Compile
